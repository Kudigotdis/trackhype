# TrackHype 3-Tier Weekly Ballot System – Complete Implementation Plan

## Overview

The **3-Tier Architecture** replaces the single-chart model with a discovery pipeline:

- **Tier 1 (Main Chart):** Top 20 from last week's votes + climbing contenders
- **Tier 2 (Contenders):** Top 50% of last week's new submissions (by upvotes)
- **Tier 3 (New Releases):** All fresh submissions this week

Users engage at all levels:
- Vote top 10 in Tier 1 (Borda 10–1 + new release boost)
- Optionally upvote/downvote in Tier 3 (discovery ballot)
- See progression from Tier 3 → Tier 2 → Tier 1 over weeks

### Date & Week Utilities (Unchanged from Previous Plan)

```javascript
function currentWeekKey() {
  const now = new Date();
  const d = new Date(Date.UTC(now.getFullYear(), now.getMonth(), now.getDate()));
  d.setUTCDate(d.getUTCDate() + 4 - (d.getUTCDay() || 7));
  const yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1));
  const weekNo = Math.ceil((((d - yearStart) / 86400000) + 1) / 7);
  return `${d.getUTCFullYear()}-W${String(weekNo).padStart(2, '0')}`;
}

function getNextSundayReset() {
  const now = new Date();
  const nextSunday = new Date(now);
  const day = now.getUTCDay();
  const daysUntilSunday = (day === 0) ? 7 : 7 - day;
  nextSunday.setUTCDate(now.getUTCDate() + daysUntilSunday);
  nextSunday.setUTCHours(0, 0, 0, 0);
  return nextSunday.getTime();
}

function weekKeyToDateRange(weekKey) {
  const [year, week] = weekKey.split('-W').map(Number);
  const jan4 = new Date(year, 0, 4);
  const weekStart = new Date(jan4);
  weekStart.setDate(jan4.getDate() - jan4.getDay() + 1);
  weekStart.setDate(weekStart.getDate() + (week - 1) * 7);
  const weekEnd = new Date(weekStart);
  weekEnd.setDate(weekStart.getDate() + 6);
  
  const monthStart = weekStart.toLocaleString('en-US', { month: 'short' });
  const dateStart = weekStart.getDate();
  const monthEnd = weekEnd.toLocaleString('en-US', { month: 'short' });
  const dateEnd = weekEnd.getDate();
  
  return monthStart === monthEnd 
    ? `${monthStart} ${dateStart}–${dateEnd}`
    : `${monthStart} ${dateStart} – ${monthEnd} ${dateEnd}`;
}

function getPreviousWeekKey(weekKey) {
  const [year, week] = weekKey.split('-W').map(Number);
  if (week === 1) {
    return `${year - 1}-W52`;
  }
  return `${year}-W${String(week - 1).padStart(2, '0')}`;
}
```

```javascript
/**
 * Compute age in days since release
 */
function computeAgeInDays(releaseDate) {
  const release = new Date(releaseDate);
  const now = new Date();
  const diffMs = now - release;
  return Math.floor(diffMs / (1000 * 60 * 60 * 24));
}

/**
 * Check if song qualifies for new-release boost (+2 points)
 * Returns true if released within last 28 days
 */
function qualifiesForNewBoost(submission) {
  return submission.ageInDays <= 28;
}
```

```javascript
 * Get user's draft Tier 1 ballot
 */
function getDraftMainBallot(chartKey) {

  const stored = localStorage.getItem(key);
  return stored ? JSON.parse(stored) : { picks: [] };

/**
 * Save draft Tier 1 ballot
 */
function saveDraftMainBallot(chartKey, picks) {
  const key = `trackhype:draft_tier1_${chartKey}`;
  const draft = {
    chartKey,
    picks,
    lastModified: Date.now()
  };
  localStorage.setItem(key, JSON.stringify(draft));
}

/**
 * Submit final Tier 1 ballot
 * Picks are from Tier 1 OR Tier 2 (contenders)
 */
function submitMainBallot(chartKey, picks) {
  if (hasSubmittedMainBallot(chartKey)) {
    throw new Error("You have already voted on the main chart this week.");
  }
  
  if (picks.length !== 10) {
    throw new Error(`Ballot must have exactly 10 picks. You have ${picks.length}.`);
  }
  
  // Validate ranks 1–10 used exactly once
  const ranks = picks.map(p => p.rank).sort((a, b) => a - b);
  if (!ranks.every((r, i) => r === i + 1)) {
    throw new Error("Picks must use ranks 1–10 exactly once.");
  }
  
  const weekKey = currentWeekKey();
  
  const ballot = {
    id: `ballot-${Date.now()}`,
    chartKey,
    weekKey,
    timestamp: Date.now(),
    picks: picks.map(p => {
      const basePoints = 11 - p.rank;  // 10 for rank 1, 1 for rank 10
      const newBoost = qualifiesForNewBoost(p) ? 2 : 0;
      return {
        rank: p.rank,
        submissionId: p.submissionId,
        title: p.title,
        artistName: p.artistName,
        points: basePoints,
        newBoost,
        finalPoints: basePoints + newBoost
      };
    }),
    submittedAt: Date.now()
  };
  
  // Append to ballot history
  const ballotKey = `trackhype:ballots_${chartKey}_${weekKey}`;
  const ballots = getMainBallots(chartKey);
  ballots.push(ballot);
  localStorage.setItem(ballotKey, JSON.stringify(ballots));
  
  // Clear draft
  localStorage.removeItem(`trackhype:draft_tier1_${chartKey}`);
  
  document.dispatchEvent(new CustomEvent('trackhype:main-ballot-submitted', { 
    detail: { chartKey, weekKey, ballot } 
  }));
  
  return ballot;
}
```

```javascript
/**
 * Submit discovery votes (for Tier 3)
 * User picks up to 5 songs to upvote and advance
 */
function submitDiscoveryVote(chartKey, upvotes, downvotes) {
  if (hasSubmittedDiscoveryVote(chartKey)) {
    throw new Error("You have already voted on the discovery pool this week.");
  }
  
  if (upvotes.length > 5) {
    throw new Error("You can upvote up to 5 songs. You selected " + upvotes.length);
  }
  
  const weekKey = currentWeekKey();
  
  const discoveryVote = {
    id: `discovery-${Date.now()}`,
    chartKey,
    weekKey,
    timestamp: Date.now(),
    upvotes,
    downvotes,
    submittedAt: Date.now()
  };
  
  // Append to discovery votes
  const discoveryKey = `trackhype:discovery_votes_${chartKey}_${weekKey}`;
  const votes = getDiscoveryVotes(chartKey);
  votes.push(discoveryVote);
  localStorage.setItem(discoveryKey, JSON.stringify(votes));
  
  // Update submission upvote counts
  const tier3 = getSubmissionsForTier(chartKey, weekKey, 3);
  
  upvotes.forEach(submissionId => {
    const sub = tier3.find(s => s.id === submissionId);
    if (sub) {
      if (!sub.discoveryUpvoters.includes("user-" + Date.now())) {
        sub.discoveryUpvotes += 1;
        sub.discoveryUpvoters.push("user-" + Date.now());
      }
    }
  });
  
  downvotes.forEach(submissionId => {
    const sub = tier3.find(s => s.id === submissionId);
    if (sub) {
      sub.discoveryDownvotes += 1;
    }
  });
  
  // Recompute upvote ratio
  tier3.forEach(sub => {
    const total = sub.discoveryUpvotes + sub.discoveryDownvotes;
    sub.upvoteRatio = total > 0 
      ? ((sub.discoveryUpvotes / total) * 100).toFixed(0) + '%'
      : '—';
  });
  
  saveSubmissionsToTier(chartKey, weekKey, 3, tier3);
  
  // Clear draft
  localStorage.removeItem(`trackhype:draft_tier3_upvotes_${chartKey}`);
  
  document.dispatchEvent(new CustomEvent('trackhype:discovery-vote-submitted', { 
    detail: { chartKey, weekKey, discoveryVote } 
  }));
  
  return discoveryVote;
}
```

```javascript
/**
 * Aggregate all main-chart ballots for Tier 1 and compute final rankings
 * Called at week boundary or on-demand for display
 */
function aggregateMainBallots(chartKey, weekKey) {
  const ballots = getMainBallots(chartKey);
  const tier1 = getSubmissionsForTier(chartKey, weekKey, 1);
  const tier2 = getSubmissionsForTier(chartKey, weekKey, 2);
  
  // Initialize score tracking for Tier 1 + Tier 2 songs only
  const songScores = {};
  
  tier1.forEach(sub => {
    songScores[sub.id] = {
      title: sub.title,
      artistName: sub.artistName,
      totalPoints: 0,
      voteCount: 0,
      isNew: qualifiesForNewBoost(sub)
    };
  });
  
  tier2.forEach(sub => {
    songScores[sub.id] = {
      title: sub.title,
      artistName: sub.artistName,
      totalPoints: 0,
      voteCount: 0,
      isNew: qualifiesForNewBoost(sub)
    };
  });
  
  // Tally votes
  ballots.forEach(ballot => {
    ballot.picks.forEach(pick => {
      if (songScores[pick.submissionId]) {
        songScores[pick.submissionId].totalPoints += pick.finalPoints;
        songScores[pick.submissionId].voteCount += 1;
      }
    });
  });
  
  // Sort and assign ranks
  const rankings = Object.entries(songScores)
    .map(([submissionId, data]) => ({
      submissionId,
      title: data.title,
      artistName: data.artistName,
      totalPoints: data.totalPoints,
      voteCount: data.voteCount,
      pickRate: ballots.length > 0 
        ? ((data.voteCount / ballots.length) * 100).toFixed(1) + '%'
        : '0%'
    }))
    .sort((a, b) => b.totalPoints - a.totalPoints)
    .slice(0, 20)
    .map((item, idx) => ({ ...item, rank: idx + 1 }));
  
  return rankings;
}

/**
 * Compute movement badges (vs. last week)
 */
function computeMovement(currentRank, lastWeekRank) {
  if (lastWeekRank === null || lastWeekRank === undefined) {
    return { label: "NEW", emoji: "🆕" };
  }
  
  const diff = lastWeekRank - currentRank;
  
  if (diff > 0) {
    return { label: `+${diff}`, emoji: "🟢" };
  } else if (diff < 0) {
    return { label: `${diff}`, emoji: "🔴" };
  } else {
    return { label: "=", emoji: "🟡" };
  }
}
```

### Weekly Reset & Promotion

```javascript
/**
 * Run at Sunday 00:00 UTC: Reset and promote tiers
 * - Archive current Tier 1 + 2 + 3 as snapshot
 * - Promote Tier 2 → becomes next week's Tier 2 (carry over)
 * - Promote top 50% of Tier 3 → becomes next week's Tier 2
 * - Top 20 from Tier 1 votes → becomes next week's Tier 1
 * - Clear Tier 3 for new submissions
 */
function executeWeeklyReset(chartKey) {
  const thisWeekKey = currentWeekKey();
  const ballots = getMainBallots(chartKey);
  const discoveryVotes = getDiscoveryVotes(chartKey);
  
  // Step 1: Aggregate and get final rankings
  const tier1Rankings = aggregateMainBallots(chartKey, thisWeekKey);
  
  // Step 2: Get current Tier 3 and sort by upvotes
  const tier3All = getSubmissionsForTier(chartKey, thisWeekKey, 3);
  const tier3Sorted = tier3All
    .sort((a, b) => b.discoveryUpvotes - a.discoveryUpvotes)
    .map((item, idx) => ({ ...item, tier3Rank: idx + 1 }));
  
  const promotionCutoff = Math.ceil(tier3Sorted.length * 0.5);  // Top 50%
  const tier3Promoted = tier3Sorted.slice(0, promotionCutoff);
  
  // Step 3: Create archive snapshot
  const previousWeekKey = getPreviousWeekKey(thisWeekKey);
  const previousTier1 = getSubmissionsForTier(chartKey, previousWeekKey, 1) || [];
  
  // Map last week's ranks for movement tracking
  const lastWeekRankMap = {};
  previousTier1.forEach(sub => {
    lastWeekRankMap[sub.id] = sub.rank || null;
  });
  
  const snapshot = {
    id: `snapshot-${chartKey}-${thisWeekKey}`,
    chartKey,
    weekKey: thisWeekKey,
    publishedAt: Date.now(),
    
    tier1_rankings: tier1Rankings.map(item => {
      const lastRank = lastWeekRankMap[item.submissionId] || null;
      const movement = computeMovement(item.rank, lastRank);
      
      return {
        rank: item.rank,
        submissionId: item.submissionId,
        title: item.title,
        artistName: item.artistName,
        totalPoints: item.totalPoints,
        ballotCount: item.voteCount,
        movement: movement.label,
        emoji: movement.emoji
      };
    }),
    
    tier3_submitted: tier3All.length,
    tier3_promoted: tier3Promoted.length,
    tier3_top_upvoters: tier3Sorted.slice(0, 10).map(item => ({
      submissionId: item.id,
      title: item.title,
      artistName: item.artistName,
      upvotes: item.discoveryUpvotes,
      upvoteRatio: item.upvoteRatio
    }))
  };
  
  // Save snapshot
  const snapshotKey = `trackhype:snapshots_${chartKey}`;
  const snapshots = getChartSnapshots(chartKey);
  snapshots.push(snapshot);
  localStorage.setItem(snapshotKey, JSON.stringify(snapshots));
  
  // Step 4: Prepare NEXT week's tiers
  const nextWeekKey = getNextWeekKey(thisWeekKey);
  
  // Next week's Tier 1 = this week's top 20
  const nextTier1 = tier1Rankings.map(item => {
    const sub = tier1All.find(s => s.id === item.submissionId) || 
               tier2All.find(s => s.id === item.submissionId);
    return {
      ...sub,
      tier: 1,
      weekKey: nextWeekKey,
      rank: item.rank,
      lastWeekRank: item.rank,
      movement: null
    };
  });
  
  // Next week's Tier 2 = top 50% from this week's Tier 3
  const nextTier2 = tier3Promoted.map((item, idx) => ({
    ...item,
    tier: 2,
    weekKey: nextWeekKey,
    discoveryUpvotes: item.discoveryUpvotes,
    movement: "Promoted from Tier 3"
  }));
  
  // Save next week's tiers
  saveSubmissionsToTier(chartKey, nextWeekKey, 1, nextTier1);
  saveSubmissionsToTier(chartKey, nextWeekKey, 2, nextTier2);
  // Tier 3 starts empty; artists submit fresh songs
  saveSubmissionsToTier(chartKey, nextWeekKey, 3, []);
  
  // Step 5: Clear current week's ballot/vote data
  localStorage.removeItem(`trackhype:ballots_${chartKey}_${thisWeekKey}`);
  localStorage.removeItem(`trackhype:discovery_votes_${chartKey}_${thisWeekKey}`);
  localStorage.removeItem(`trackhype:draft_tier1_${chartKey}`);
  localStorage.removeItem(`trackhype:draft_tier3_upvotes_${chartKey}`);
  
  document.dispatchEvent(new CustomEvent('trackhype:weekly-reset', { 
    detail: { chartKey, weekKey: thisWeekKey, snapshot } 
  }));
  
  return snapshot;
}

/**
 * Get next week's ISO key (for week math)
 */
function getNextWeekKey(weekKey) {
  const [year, week] = weekKey.split('-W').map(Number);
  if (week === 52) {
    return `${year + 1}-W01`;
  }
  return `${year}-W${String(week + 1).padStart(2, '0')}`;
}
```

### Archive & Historical View

```javascript
/**
 * Load all snapshots for a chart
 */
function getChartSnapshots(chartKey) {
  const key = `trackhype:snapshots_${chartKey}`;
  const stored = localStorage.getItem(key);
  return stored ? JSON.parse(stored) : [];
}

/**
 * Get snapshot for a specific week
 */
function getSnapshotForWeek(chartKey, weekKey) {
  const snapshots = getChartSnapshots(chartKey);
  return snapshots.find(s => s.weekKey === weekKey);
}

/**
 * Get available weeks for archive dropdown
 */
function getAvailableWeeks(chartKey) {
  const snapshots = getChartSnapshots(chartKey);
  return snapshots
    .map(s => ({
      weekKey: s.weekKey,
      label: `Week ${s.weekKey.split('W')[1]} (${weekKeyToDateRange(s.weekKey)})`
    }))
    .reverse()
    .slice(0, 13);  // Last 13 weeks (3 months)
}
```

```javascript
function refreshTier1View() {

function renderRankSlotsForBallot() {
  const container = document.getElementById('current-picks');
  let html = '';
  
  for (let rank = 1; rank <= 10; rank++) {
    const pick = uiState.mainBallotPicks.find(p => p.rank === rank);
    html += `
      <div class="rank-slot ${pick ? 'filled' : ''}" onclick="editRankInBallot(${rank})">
        <div class="rank-number">#${rank}</div>
        ${pick ? `<div class="rank-song">${pick.title}</div>` : '<div class="rank-song">Tap to add</div>'}
      </div>

  }
  
  document.getElementById('current-picks').innerHTML = html;
  updateBallotProgress();
}

function addSongToBallot(submissionId, title, artistName) {
  // Find next available rank
  for (let rank = 1; rank <= 10; rank++) {
    if (!uiState.mainBallotPicks.find(p => p.rank === rank)) {
      uiState.mainBallotPicks.push({
        submissionId,
        rank,
        title,
        artistName
      });
      break;
    }
  }
  
  saveDraftMainBallot(uiState.chartKey, uiState.mainBallotPicks);
  renderAvailableSongsForBallot();
  renderRankSlotsForBallot();
}

function editRankInBallot(rank) {
  const pick = uiState.mainBallotPicks.find(p => p.rank === rank);
  
  if (!pick) return;
  
  // Allow user to move to different rank or remove
  const newRank = prompt(
    `Move "${pick.title}" to a different rank (1-10), or press Cancel to remove:`,
    rank
  );
  
  if (newRank === null) {
    // Remove
    uiState.mainBallotPicks = uiState.mainBallotPicks.filter(p => p.submissionId !== pick.submissionId);
  } else {
    const newRankNum = parseInt(newRank, 10);
    if (newRankNum >= 1 && newRankNum <= 10 && newRankNum !== rank) {
      // Swap ranks if needed
      const other = uiState.mainBallotPicks.find(p => p.rank === newRankNum);
      if (other) {
        other.rank = rank;
      }
      pick.rank = newRankNum;
    }
  }
  
  saveDraftMainBallot(uiState.chartKey, uiState.mainBallotPicks);
  renderAvailableSongsForBallot();
  renderRankSlotsForBallot();
}

function updateBallotProgress() {

  document.getElementById('main-progress-fill').style.width = `${(count / 10) * 100}%`;
  document.getElementById('submit-main-ballot-btn').disabled = count !== 10;
}

function toggleDiscoveryUpvote(submissionId, isUpvote) {
  const isAlreadyUpvoted = uiState.discoveryUpvotes.includes(submissionId);
  
  if (isAlreadyUpvoted) {
    uiState.discoveryUpvotes = uiState.discoveryUpvotes.filter(id => id !== submissionId);
  } else {
    if (uiState.discoveryUpvotes.length >= 5) {
      alert('You can upvote up to 5 songs. Remove one to add another.');
      return;
    }
    uiState.discoveryUpvotes.push(submissionId);
    // Remove from downvotes if present
    uiState.discoveryDownvotes = uiState.discoveryDownvotes.filter(id => id !== submissionId);
  }
  
  saveDraftDiscoveryVotes(uiState.chartKey, uiState.discoveryUpvotes, uiState.discoveryDownvotes);
  renderSongsForDiscovery();
  renderDiscoverySelection();
}

function toggleDiscoveryDownvote(submissionId) {
  const isAlreadyDownvoted = uiState.discoveryDownvotes.includes(submissionId);
  
  if (isAlreadyDownvoted) {
    uiState.discoveryDownvotes = uiState.discoveryDownvotes.filter(id => id !== submissionId);
  } else {
    uiState.discoveryDownvotes.push(submissionId);
    // Remove from upvotes if present
    uiState.discoveryUpvotes = uiState.discoveryUpvotes.filter(id => id !== submissionId);
  }
  
  saveDraftDiscoveryVotes(uiState.chartKey, uiState.discoveryUpvotes, uiState.discoveryDownvotes);
  renderSongsForDiscovery();
  renderDiscoverySelection();
}

function updateCountdown() {
  const resetTime = getNextSundayReset();
  
  const timer = setInterval(() => {
    const now = Date.now();
    const diff = resetTime - now;
    
    if (diff <= 0) {
      clearInterval(timer);
      document.getElementById('countdown-text').textContent = 
        "Chart reset! New votes available now.";
      return;
    }
    
    const days = Math.floor(diff / (1000 * 60 * 60 * 24));
    const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
    const minutes = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
    
    document.getElementById('countdown-text').textContent = 
      `Next reset in: ${days}d ${hours}h ${minutes}m`;
  }, 1000);
}
```
